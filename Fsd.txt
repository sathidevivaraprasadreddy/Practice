// Function Declaration
function add(a, b) {
    return a + b;
}

// Function Definition / Function Expression
let sub = function(a, b) {
    return a - b;
};

// Arrow Function
let mul = (a, b) => a * b;

console.log("Addition:", add(10, 5));
console.log("Subtraction:", sub(10, 5));
console.log("Multiplication:", mul(10, 5));
